# Hosts a match through Steam, alone, and waits for a joiner (see
# steam_find.txt) to find it, join, and leave. Run with tools/play.ps1
# -Steam; the Steam client must be running:
#
#     powershell -ExecutionPolicy Bypass -File tools/play.ps1 -Steam -Script tools/play/steam_host.txt -ClientScript tools/play/steam_find.txt -ClientAlone -Tag steam
