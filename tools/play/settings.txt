# Looks at the front end and its pages: the main page over the menu's tile
# of cheese, the credits, and each tab of the settings (the controls with the
# mouse and a gamepad, and at the smallest window); then hosts a match from
# the front end, as its button does, into the lobby's usual rules. Run with
# tools/play.ps1 -Lobby (alone, on the front end).
wait 8
cmd r.SetRes 1600x900w
wait 1
shot settings_main
cmd CheeseStrikePage Credits
wait 1
shot settings_credits
cmd CheeseStrikePage Settings Controls
wait 1
shot settings_controls
device gamepad
wait 1
shot settings_controls_pad
device mouse
cmd r.SetRes 1280x720w
wait 1
shot settings_controls_720
cmd r.SetRes 1600x900w
cmd CheeseStrikePage Settings Aiming
wait 1
shot settings_aiming
cmd CheeseStrikePage Settings View
wait 1
shot settings_view
cmd CheeseStrikePage Settings Sound
wait 1
shot settings_sound
cmd CheeseStrikePage Settings Graphics
wait 1
shot settings_graphics
cmd CheeseStrikePage Main
wait 1
key Enter
wait 1
shot settings_hosting
wait 12
shot settings_hosted_lobby
quit
